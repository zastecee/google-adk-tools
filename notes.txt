SELECT /*+PARALLEL(15)*/
 TRUNC(DC_DATE, 'MM') PP_MONTH,                         -- "2026-04-01 00:00:00.0" -> GROUP +
 BO.GET_WEEK_NUM(DC_DATE) WEEK_NUM,                     -- "W3" -> GROUP +
 BO.GET_SAME_DAY(DC_DATE) SAME_DAY,                     -- "2026-08-15 00:00:00.0" -> GROUP
 TO_CHAR(DC_DATE, 'DY') WEEK_DAY,                       -- "SAT" -> GROUP +
 BO.GET_DATE_MTD(DC_DATE) DATE_MTD,                     -- "PM1TD"  -> GROUP
 D.DC_DATE,                                             -- "2026-04-18 00:00:00.0" -> GROUP +
 D.EVENT_DESCRIPTION,                                   -- "30 Days Smartphone Users" -> GROUP +
 SUM(D.DC_PREPAD) DC_PREPAD,                            -- "4265"
 SUM(D.DC_HYBRID) DC_HYBRID,                            -- "98"
 SUM(D.DC_TOTAL) DC_TOTAL,                              -- "4450"
 D.DC_PROVINCE,                                         -- "Manica" -> GROUP +
 D.DC_SEGMENT,                                          -- "HIGH" -> GROUP +
 NULL,                                                  -- null
 TO_CHAR(TRUNC(DC_DATE, 'mm'), 'MM') MMONTH,            -- "04"
 TO_CHAR(TRUNC(DC_DATE, 'dd'), 'DD') DDATE,             -- "18"
 TO_CHAR(TRUNC(DC_DATE, 'yyyy'), 'YYYY') YYEAR          -- "2026"
  FROM BO.V_DRA_DAILY_CUSTOMERS_SELECT D
 WHERE DC_DATE >= ADD_MONTHS(TRUNC(SYSDATE, 'MM'), -5)
   AND DC_DATE < TRUNC(SYSDATE)
 GROUP BY TRUNC(DC_DATE, 'MM'),                         -- "2026-04-01 00:00:00.0"      
          BO.GET_WEEK_NUM(DC_DATE),                     -- "W3"
          BO.GET_SAME_DAY(DC_DATE),                     -- "2026-08-15 00:00:00.0"
          TO_CHAR(DC_DATE, 'DY'),                       -- "SAT"                        -- SAT
          BO.GET_DATE_MTD(DC_DATE),                     -- "PM1TD"                      -- PM1TD
          D.DC_DATE,                                    -- "2026-04-18 00:00:00.0"      --
          D.EVENT_DESCRIPTION,                          -- "30 Days Smartphone Users"   --
          D.DC_PROVINCE,                                -- "Manica"                     -- Manica
          D.DC_SEGMENT;                                 -- "HIGH"                       -- HIGH
